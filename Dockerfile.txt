# Myanmar Recap Studio — HuggingFace Spaces ready
FROM python:3.10-slim

# ffmpeg for video/audio processing (+ Myanmar font for title cards if available)
RUN apt-get update && apt-get install -y --no-install-recommends \
        ffmpeg \
    && (apt-get install -y --no-install-recommends fonts-sil-padauk || true) \
    && rm -rf /var/lib/apt/lists/*

WORKDIR /app

COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

COPY . .

EXPOSE 7860

CMD ["python", "app.py"]
